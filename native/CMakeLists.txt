cmake_minimum_required(VERSION 3.22)
project(broadband_native LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(broadband_native SHARED probes/probe_engine.c jni/probes_jni.c advanced/packet_engine.c jni/packets_jni.c jni/ethernet_jni.c)
target_include_directories(broadband_native PRIVATE probes advanced)
target_compile_options(broadband_native PRIVATE -Wall -Wextra -Werror -O2)
target_link_libraries(broadband_native android log)
target_link_options(broadband_native PRIVATE -Wl,-z,max-page-size=16384)
